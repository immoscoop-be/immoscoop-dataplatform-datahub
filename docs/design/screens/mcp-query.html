The SQL, written by Claude 
with weekly as (
  select week_start, lead_source, lead_count
  from `immoscoop-dataplatform-398214.dbt_prod_stacks.weekly_lead_volume`
  where week_start >= date_sub(date_trunc(current_date(), week(monday)), interval 4 week)
    and week_start < date_trunc(current_date(), week(monday))  where week_start >= date_sub(date_trunc(current_date(), week(monday)), interval 26 week)
    and week_start < date_trunc(current_date(), week(monday))  where week_start >= '2024-01-01'
    and week_start < date_trunc(current_date(), week(monday))
)
select w.week_start,
       sum(if(s.is_partner, w.lead_count, 0)) as partner_leads,
       sum(w.lead_count) as all_leads,
       round(100 * sum(if(s.is_partner, w.lead_count, 0)) / sum(w.lead_count), 1) as partner_share
from weekly as w
join `immoscoop-dataplatform-398214.dbt_prod_stacks.lead_sources` as s using (lead_source)
group by w.week_start
order by w.week_start desc
