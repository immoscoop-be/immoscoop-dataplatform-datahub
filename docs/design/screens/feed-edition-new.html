1-- Prices of sales per street and per municipality, per month and type of property
2with
3sales as (select street_id, street_name, municipality_id, municipality_name, property_type,
4 date_trunc(sale_date, month) as period_month, price
5 from `dbt_prod_listings.sales_prices` inner join `dbt_prod_geo.streets_and_municipalities` using (street_id)),
6agencies as (select municipality_id, count(distinct company_id) as active_agencies
7 from `dbt_prod_agencies.agencies` where is_active group by municipality_id),
8per_area as (select 'street' as area_level, cast(street_id as string) as area_id, street_name as area_name,
9 property_type, period_month, count(*) as n_sales, avg(price) as price_mean,
10 approx_quantiles(price, 4)[offset(2)] as price_median, approx_quantiles(price, 4)[offset(1)] as price_p25,
11 approx_quantiles(price, 4)[offset(3)] as price_p75,
12 stddev_samp(price) as price_sd
13 from sales group by 1, 2, 3, 4, 5
14 -- … the same per municipality, union all
15)
16select per_area.*, agencies.active_agencies from per_area
17left join agencies on per_area.area_id = cast(agencies.municipality_id as string)
